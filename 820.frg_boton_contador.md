# Fragmento FrgBotonContador (Android - Java)

Fragmento que encapsula un `Button` que solo se puede clicar un número determinado de veces.

Se tendrán en cuenta las características que puedan surgir en su uso, como la posibilidad de cancelar un clic en un momento inadecuado de su clase contenedora.

## Interfaz del fragmento **frg_boton_contador.xml**

```xml
<?xml version="1.0" encoding="utf-8"?>
<FrameLayout xmlns:android="http://schemas.android.com/apk/res/android"
    android:layout_width="wrap_content"
    android:layout_height="wrap_content">

    <Button
        android:id="@+id/boton"
        android:layout_width="wrap_content"
        android:layout_height="wrap_content" />
</FrameLayout>
```

## Código del fragmento  **FrgBotonContador.java**

```java
import android.os.Bundle;
import androidx.annotation.NonNull;
import androidx.annotation.Nullable;
import androidx.fragment.app.Fragment;
import android.view.LayoutInflater;
import android.view.View;
import android.view.ViewGroup;
import android.widget.Button;

public class FrgBotonContador extends Fragment {

    public interface OnBotonContadorListener {
        boolean onClicBoton(FrgBotonContador frgBotonContador, int numClic, int clicsRestantes);
        default void onUltimoClic(FrgBotonContador frgBotonContador) {}
    }
    private OnBotonContadorListener listener;

    private int maxClics=0,numClics=0;
    private String texto="";

    private Button boton;

    private static final String ARG_MAX_CLICS = "maxClics";
    private static final String ARG_NUM_CLICS = "numClics";
    private static final String ARG_TEXTO = "texto";


    public FrgBotonContador() {} // constructor vacío obligatorio. Ver documentación.

    @Override
    public View onCreateView(@NonNull LayoutInflater inflater, @Nullable ViewGroup container, @Nullable Bundle savedInstanceState) {
        return inflater.inflate(R.layout.frg_boton_contador, container, false);
    }

    @Override
    public void onViewCreated(@NonNull View view, @Nullable Bundle savedInstanceState) {
        super.onViewCreated(view, savedInstanceState);

        if(savedInstanceState!=null) {
            maxClics=savedInstanceState.getInt(ARG_MAX_CLICS,0);
            numClics=savedInstanceState.getInt(ARG_NUM_CLICS,0);
            texto=savedInstanceState.getString(ARG_TEXTO,"");
        }

        boton = view.findViewById(R.id.boton);
        boton.setOnClickListener(v -> procesarClic());
        actualizarVista();
    }

    @Override
    public void onSaveInstanceState(@NonNull Bundle outState) {
        super.onSaveInstanceState(outState);
        outState.putInt(ARG_MAX_CLICS,maxClics);
        outState.putInt(ARG_NUM_CLICS,numClics);
        outState.putString(ARG_TEXTO,texto);
    }

    @Override
    public void onDestroyView() {
        super.onDestroyView();
        boton = null;
    }

    private void procesarClic() {
        if (!quedanClics()) return;
        boolean clicAceptado=true;
        if (listener!=null) clicAceptado=listener.onClicBoton(this,numClics+1,maxClics-numClics-1);
        if (!clicAceptado) return;
        numClics++;
        actualizarVista();
        if (!quedanClics() && listener!=null) listener.onUltimoClic(this);
    }

    private void actualizarVista() {
        if (boton==null) return;
        boton.setEnabled(quedanClics());
        boton.setText(String.format("%s %d/%d", texto, numClics, maxClics));
    }

    public void setOnBotonLimitadoListener(OnBotonContadorListener listener) {
        this.listener = listener;
    }

    public void setMaxClics(int maxClics) {
        if (maxClics < 0) throw new IllegalArgumentException("El número máximo no puede ser negativo");
        this.maxClics = maxClics;
        if (numClics > maxClics) numClics = maxClics;
        actualizarVista();
    }

    public void setText(String texto) {
        this.texto=texto;
        actualizarVista();
    }

    public boolean quedanClics()       { return numClics < maxClics; }
}
```

## Código de la actividad  **MainActivity.java**

```java
package com.example.fragmentobotoncontador;

import android.os.Bundle;
import android.widget.Toast;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;
import androidx.core.graphics.Insets;
import androidx.core.view.ViewCompat;
import androidx.core.view.WindowInsetsCompat;
import androidx.fragment.app.FragmentManager;

public class MainActivity extends AppCompatActivity implements FrgBotonContador.OnBotonContadorListener {
    public static final int MAX_CLICS = 3;
    public static final int CLICS_DE_REGALO=2;
    private boolean regaloEntregado=false;

    FrgBotonContador frgBotonContador;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);
        ViewCompat.setOnApplyWindowInsetsListener(findViewById(R.id.main), (v, insets) -> {
            Insets systemBars = insets.getInsets(WindowInsetsCompat.Type.systemBars());
            v.setPadding(systemBars.left, systemBars.top, systemBars.right, systemBars.bottom);
            return insets;
        });

        FragmentManager fm=getSupportFragmentManager();
        frgBotonContador= (FrgBotonContador) fm.findFragmentById(R.id.frgBotonContador);
        frgBotonContador.setMaxClics(MAX_CLICS);
        frgBotonContador.setText("Prueba");
        frgBotonContador.setOnBotonLimitadoListener(this);
    }

    @Override
    public boolean onClicBoton(FrgBotonContador frgBotonContador, int numClic, int clicsRestantes) {
        Toast.makeText(this,"Clic "+numClic+" de  "+(numClic+clicsRestantes),Toast.LENGTH_SHORT).show();
        return true;
    }

    @Override
    public void onUltimoClic(FrgBotonContador frgBotonContador) {
        Toast.makeText(this,"Y fin",Toast.LENGTH_SHORT).show();
        if(!regaloEntregado) {
            Toast.makeText(this,"Y ahora "+CLICS_DE_REGALO+" de regalo!",Toast.LENGTH_SHORT).show();
            frgBotonContador.setMaxClics(MAX_CLICS+CLICS_DE_REGALO);
            regaloEntregado=true;
        }
    }
}
```
